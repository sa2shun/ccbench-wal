\section{Proposal: Ayame}
\label{sec:design}

A frontier $F$ is a vector of per-shard log positions.  $F \fjoin F'$
denotes the element-wise maximum, and $F \le F'$ holds iff $F[i] \le F'[i]$
for every shard $i$.  $D$ denotes the durable vector, where $D[i]$ is the
durable position of shard $i$.

\begin{algorithm}[t]
\small
\caption{Dependency collection.}
\label{alg:ayame-dep}
\begin{algorithmic}[1]
\Procedure{ReadVersion}{$T, v$}
  \State add $v$ to $T$'s read set
  \If{$T$ is known read-only}
    \State \Return \Comment{SI read-only fast path}
  \EndIf
  \State $T.\dep \gets T.\dep \fjoin v.\wfr$
        \label{ln:dep-read}
\EndProcedure
\Statex
\Procedure{OverwriteVersion}{$T, v$}
  \State $T.\dep \gets T.\dep \fjoin v.\wfr \fjoin v.\rfr$
        \Comment{writer and reader frontiers}
        \label{ln:dep-write}
  \State add $v$ to $T$'s write set
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\small
\caption{Worker-side commit protocol.}
\label{alg:ayame-commit}
\begin{algorithmic}[1]
\Require transaction $T$ with dependency frontier $T.\dep$
\If{\Call{SsnValidate}{$T$} fails}
  \State \textbf{abort} $T$ and \Return
\EndIf
\If{$T$ is read-only}
  \If{$T.\dep \le D$}
    \State \Call{Ack}{$T$}; \Return
  \EndIf
  \State \Call{Register}{$T.\dep, T$}; \Return
        \Comment{Algorithm~\ref{alg:ayame-ack}}
\EndIf
\State $T.c \gets$ commit timestamp assigned by ERMIA/SSN
       \Comment{logical LSN}
       \label{ln:cstamp}
\State $s \gets$ WAL shard owned by this worker
\State $r \gets$ \Call{BuildRecord}{$T.c, T.\dep, T.\mathit{write\_set}$}
\State $T.\lseq \gets$ \Call{Enqueue}{$\mathit{wal}[s], r$}
       \Comment{append only; no flush, no wait}
       \label{ln:enqueue}
\State $F \gets T.\dep$; \quad $F[s] \gets \max(F[s], T.\lseq)$
       \Comment{closed frontier}
       \label{ln:closed}
\ForAll{versions $v$ created by $T$}
  \State $v.\wfr \gets F$
        \label{ln:pub-w}
\EndFor
\ForAll{versions $v$ read by read-write $T$}
  \State $v.\rfr \gets v.\rfr \fjoin F$
        \label{ln:pub-r}
\EndFor
\State \Call{Register}{$F, T$}
       \Comment{durable ack deferred to the committer}
       \label{ln:register}
\State \Return logical commit
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\small
\caption{Committer-side durable acknowledgment.}
\label{alg:ayame-ack}
\begin{algorithmic}[1]
\Statex \textbf{State:} durable vector $D$ and per-shard waitlists $W[i]$
\Procedure{Register}{$F, T$}
  \If{$F \le D$}
    \State \Call{Ack}{$T$}; \Return
        \Comment{own log and dependencies are durable}
        \label{ln:dur-check}
  \EndIf
  \State $T.\mathit{remaining} \gets 0$
  \ForAll{shards $i$ such that $F[i] > D[i]$}
    \State $T.\mathit{remaining} \gets T.\mathit{remaining} + 1$
    \State insert $(F[i], T)$ into $W[i]$
          \label{ln:park}
  \EndFor
\EndProcedure
\Statex
\Procedure{OnDurableAdvance}{$i, d$}
        \Comment{called by flusher $i$ after \texttt{fdatasync}}
  \State $D[i] \gets d$
  \While{$W[i]$ is not empty and \Call{MinNeed}{$W[i]$} $\le d$}
    \State $(\mathit{need}, T) \gets$ \Call{PopMin}{$W[i]$}
    \State $T.\mathit{remaining} \gets T.\mathit{remaining} - 1$
    \If{$T.\mathit{remaining} = 0$}
      \State \Call{Ack}{$T$}
          \label{ln:ack-ready}
    \EndIf
  \EndWhile
\EndProcedure
\end{algorithmic}
\end{algorithm}
